def sample_gradient(h=360, w=480):
    """A smooth RGB gradient. Great for watching interpolation do its job."""
    yy, xx = np.mgrid[0:h, 0:w]
    rgb = np.zeros((h, w, 3), dtype=np.uint8)
    rgb[..., 0] = (255 * xx / (w - 1)).astype(np.uint8)          # R grows → right
    rgb[..., 1] = (255 * yy / (h - 1)).astype(np.uint8)          # G grows → down
    rgb[..., 2] = (128 + 127 * np.sin(np.pi * xx / w)).astype(np.uint8)
    return rgb


def sample_checker(h=360, w=480, cell=40):
    """A checkerboard. Rotation and shear become very visible."""
    yy, xx = np.indices((h, w))
    on = ((xx // cell) + (yy // cell)) % 2 == 0
    rgb = np.zeros((h, w, 3), dtype=np.uint8)
    rgb[on] = (235, 235, 235)
    rgb[~on] = (24, 40, 66)
    return rgb


def sample_shapes(h=360, w=480):
    """A tiny geometric 'scene' (sky, sun, house, tree)."""
    yy, xx = np.indices((h, w))
    rgb = np.zeros((h, w, 3), dtype=np.uint8)
    rgb[..., 0], rgb[..., 1], rgb[..., 2] = 40, 90, 150          # sky
    sun = (xx - 360) ** 2 + (yy - 80) ** 2 < 2500
    rgb[sun] = (255, 210, 70)
    house = (xx > 90) & (xx < 300) & (yy > 190) & (yy < 330)
    rgb[house] = (235, 230, 215)
    door = (xx > 130) & (xx < 180) & (yy > 230) & (yy < 330)
    rgb[door] = (90, 70, 60)
    roof = (yy > 130) & (yy < 220) & (xx > 90) & (xx < 300) & \
           (yy > 130 + 90 * np.abs((xx - 90) / 210 - 0.5) * 2)
    rgb[roof] = (200, 70, 60)
    crown = (xx - 380) ** 2 + (yy - 260) ** 2 < 1600
    rgb[crown] = (60, 160, 70)
    trunk = (xx > 374) & (xx < 386) & (yy > 280) & (yy < 330)
    rgb[trunk] = (110, 80, 50)
    return rgb

def to_grayscale(arr):
    """Luminance weights: I = 0.299·R + 0.587·G + 0.114·B   (the Rec. 601 weights)"""
    if arr.ndim == 2:
        return arr.astype(np.float64)
    return (0.299 * arr[..., 0] + 0.587 * arr[..., 1] + 0.114 * arr[..., 2]).astype(
        np.uint8
    )


# =============================================================================
# STEP 6 — transformation matrices (geometric, act on COORDINATES)
# =============================================================================
def rotation_matrix(deg):
    """R(θ) = [[cosθ, -sinθ], [sinθ, cosθ]]  (degrees; anti-clockwise in math coords)"""
    t = np.radians(deg)
    c, s = np.cos(t), np.sin(t)
    return np.array([[c, -s], [s, c]])


def scaling_matrix(sx, sy):
    return np.array([[sx, 0.0], [0.0, sy]])


def shear_matrix(k, axis="x"):
    """Horizontal shear [[1, k],[0, 1]] or vertical shear [[1, 0],[k, 1]]."""
    if axis == "x":
        return np.array([[1.0, k], [0.0, 1.0]])
    return np.array([[1.0, 0.0], [k, 1.0]])


def reflection_matrix(axis="x"):
    if axis == "x":
        return np.array([[1.0, 0.0], [0.0, -1.0]])
    if axis == "y":
        return np.array([[-1.0, 0.0], [0.0, 1.0]])
    return np.array([[0.0, 1.0], [1.0, 0.0]])          # across y = x


# =============================================================================
# STEP 7 — transform ONE point
# =============================================================================
def transform_point(p, A):
    """Matrix-vector multiplication p′ = A·p, written out component by component."""
    x, y = p
    a, b, c, d = A.ravel()
    xp = a * x + b * y
    yp = c * x + d * y
    print(f"p′  = A·p")
    print(f"x′  = a·x + b·y = {a}·{x} + {b}·{y} = {xp}")
    print(f"y′  = c·x + d·y = {c}·{x} + {d}·{y} = {yp}")
    return np.array([xp, yp])


# =============================================================================
# STEP 8/9 — transform an IMAGE via inverse mapping (the heart of the lab)
# =============================================================================
def _inverse_map_2x2(A):
    """Closed-form inverse of a 2×2 matrix (educational version of np.linalg.inv)."""
    a, b, c, d = A.ravel()
    det = a * d - b * c
    if abs(det) < 1e-9:
        raise ValueError(f"Matrix is singular (det = {det}). Inverse does not exist.")
    return np.array([[d, -b], [-c, a]]) / det


def transform_image(image_array, A, interpolation="nearest"):
    """Apply the 2×2 geometric matrix A to an image, using INVERSE MAPPING.

    Why inverse mapping?
      * Forward mapping (push every input pixel to its new spot) leaves
        HOLES when a transform stretches the image.
      * Inverse mapping instead asks, for every OUTPUT pixel:
            "which input pixel should I read my colour from?"
        p_in = A⁻¹ · p_out
      * If A⁻¹ p_out is not an integer pixel position we must INTERPOLATE:
            nearest  → round to the closest pixel (chunky, no new colours)
            bilinear → blend the 4 surrounding pixels (smooth)

    Rotation / scaling / shear are all around the image CENTRE.
    """
    h, w = image_array.shape[:2]
    cx, cy = w / 2.0, h / 2.0                                 # image centre (continuous)

    # --- bounding box from pixel BOUNDARY corners (prevents cropping) --------
    corners = np.array([[-cx, -cy], [w - cx, -cy],
                        [w - cx, h - cy], [-cx, h - cy]])
    tcor = corners @ A.T
    minx, maxx = tcor[:, 0].min(), tcor[:, 0].max()
    miny, maxy = tcor[:, 1].min(), tcor[:, 1].max()
    nw, nh = int(np.ceil(maxx - minx)), int(np.ceil(maxy - miny))

    # --- for every output pixel, find its source position --------------------
    # output pixel (ox, oy) has its centre at (minx + ox + 0.5, miny + oy + 0.5)
    ox = np.arange(nw)
    oy = np.arange(nh)
    U, V = np.meshgrid(minx + ox + 0.5, miny + oy + 0.5)      # shape (nh, nw)
    Ainv = _inverse_map_2x2(A)
    # U, V are already centered coords; inverse-map them to the input frame:
    inx = Ainv[0, 0] * U + Ainv[0, 1] * V + cx                # continuous input x
    iny = Ainv[1, 0] * U + Ainv[1, 1] * V + cy                # continuous input y
    # input pixel i has its centre at i + 0.5, so convert to index space:
    ix = inx - 0.5                                            # fractional source x
    iy = iny - 0.5                                            # fractional source y

    if interpolation == "nearest":
        ixi = np.clip(np.round(ix).astype(int), 0, w - 1)
        iyi = np.clip(np.round(iy).astype(int), 0, h - 1)
        out = image_array[iyi, ixi]
    elif interpolation == "bilinear":
        x0 = np.floor(ix).astype(int)
        y0 = np.floor(iy).astype(int)
        fx = ix - x0                                          # fraction into cell
        fy = iy - y0
        x0c = np.clip(x0, 0, w - 1)
        x1c = np.clip(x0 + 1, 0, w - 1)
        y0c = np.clip(y0, 0, h - 1)
        y1c = np.clip(y0 + 1, 0, h - 1)
        tl = image_array[y0c, x0c].astype(np.float64)
        tr = image_array[y0c, x1c].astype(np.float64)
        bl = image_array[y1c, x0c].astype(np.float64)
        br = image_array[y1c, x1c].astype(np.float64)
        top = tl * (1 - fx)[..., None] + tr * fx[..., None]
        bot = bl * (1 - fx)[..., None] + br * fx[..., None]
        out = (top * (1 - fy)[..., None] + bot * fy[..., None]).round().astype(
            image_array.dtype)
    else:
        raise ValueError("interpolation must be 'nearest' or 'bilinear'")
    return out, (nw, nh)

def plot_vectors(vectors, labels=None, title="Vectors on a plane",

# STEP 16 — determinant
# =============================================================================
def det2x2(A):
    """det(A) = ad − bc  (the signed area scaling factor)."""
    a, b, c, d = A.ravel()
    return a * d - b * c


# =============================================================================
# STEP 17 — inverse
# =============================================================================
def inverse_matrix(A, tol=1e-9):
    """A⁻¹, or None if det(A) ≈ 0 (singular ⇒ not invertible)."""
    det = det2x2(A)
    if abs(det) < tol:
        return None
    a, b, c, d = A.ravel()
    return np.array([[d, -b], [-c, a]]) / det

# =============================================================================
# BUILD A MINI GOOGLE — PageRank & Eigenvectors Lab (Python / NumPy edition)
# -----------------------------------------------------------------------------
# The *coding* companion to the web simulation. Every mathematical step is
# written out explicitly so you can see exactly what happens:
#
#   WEB PAGES → LINKS → DIRECTED GRAPH → ADJACENCY MATRIX → TRANSITION MATRIX
#   → INITIAL RANK → ITERATION → CONVERGENCE → STABLE VECTOR → EIGENVECTOR
#   → PAGERANK
#
# We deliberately do NOT use networkx.pagerank() — you will see the algorithm
# as raw matrix-vector multiplication.
#
# CONVENTION (kept consistent everywhere in this file):
#   * pages are indexed 0..n-1; links are (from, to) pairs, "from -> to".
#   * the ADJACENCY matrix A has A[i][j] = 1  iff page j links to page i
#     (column j = out-links of page j).
#   * the TRANSITION matrix P is COLUMN-STOCHASTIC:
#         P[i][j] = 1 / outdeg(j)   for each link j -> i
#     so that   r_{k+1} = P @ r_k   (columns describe where probability
#     flows FROM page j).
#   * a page with NO outgoing links is a "dangling" page. With
#     fix_dangling=True its column becomes uniform 1/n (probability
#     redistributed to every page).
#   * the GOOGLE matrix  G = alpha * P + (1 - alpha) / n * ones(n, n)
#     adds teleportation ("random jump") so no rank can get trapped.
#
# Author: generated for the PageRank & Eigenvectors course orientation
# =============================================================================

import numpy as np

# =============================================================================
# STEP 1 — create the network (a tiny hand-built "web")

def sample_network():
    """The example from the lecture: A, B, C, D, E with hand-picked links.

    A -> B, A -> C, B -> C, C -> A, D -> C, D -> A, E -> D
    """
    pages = ["A", "B", "C", "D", "E"]
    links = [(0, 1), (0, 2), (1, 2), (2, 0), (3, 2), (3, 0), (4, 3)]
    return pages, links


def draw_network(pages, links, node_sizes=None, title="The mini web",
                 figsize=(8, 6), pos=None):
    """Draw a directed graph with NetworkX. node_sizes scales pages by rank."""
    G = nx.DiGraph()
    G.add_nodes_from(range(len(pages)))
    G.add_edges_from(links)
    labels = {i: pages[i] for i in range(len(pages))}
    if pos is None:
        pos = nx.spring_layout(G, seed=7, k=1.2)
    plt.figure(figsize=figsize)
    if node_sizes is None:
        node_sizes = [800] * len(pages)
    nx.draw_networkx(G, pos, labels=labels, node_color="#22d3ee",
                     node_size=node_sizes, font_size=12, font_color="#06121a",
                     font_weight="bold", arrows=True, arrowstyle="-|>",
                     arrowsize=18, edge_color="#8fa3c0", width=2)
    plt.title(title)
    plt.axis("off")
    plt.tight_layout()
    plt.show()
    return pos


# =============================================================================
# STEP 2 — adjacency matrix
# =============================================================================
def adjacency_matrix(n_pages, links):
    """A[i][j] = 1 iff page j links to page i. (Column j = out-links of j.)"""
    A = np.zeros((n_pages, n_pages), dtype=float)
    for (j, i) in links:                      # j -> i
        A[i, j] = 1.0
    return A


# =============================================================================
# STEP 3 — transition matrix (column-stochastic)
# =============================================================================
def transition_matrix(A, fix_dangling=True):
    """Turn adjacency into the column-stochastic transition matrix P.

    P[i][j] = A[i][j] / outdeg(j). Columns sum to 1, except dangling pages
    (outdeg 0): if fix_dangling is True their column becomes uniform 1/n,
    otherwise it stays zero (which makes probability leak).
    """
    n = A.shape[0]
    outdeg = A.sum(axis=0)                    # out-degrees of each page
    P = np.zeros_like(A)
    for j in range(n):
        if outdeg[j] == 0:
            if fix_dangling:
                P[:, j] = 1.0 / n             # redistribute uniformly
        else:
            P[:, j] = A[:, j] / outdeg[j]     # split equally among out-links
    return P


def check_stochastic(P, label="P"):
    """Verify every column sums to 1 (this is the stochasticity test)."""
    colsums = P.sum(axis=0)
    print(f"column sums of {label}: {np.round(colsums, 6)}")
    assert np.allclose(colsums, 1.0), f"{label} is NOT column-stochastic!"
    print(f"✓ {label} is column-stochastic.")
    return colsums


# =============================================================================
# STEP 4 — initial rank vector
# =============================================================================
def initial_rank(n):
    """r0 = [1/n, 1/n, ..., 1/n]  — nobody is important yet."""
    return np.ones(n) / n


# =============================================================================
# STEPS 5 & 6 — iterate the power method by hand
# =============================================================================
def power_iteration(M, r0=None, alpha=None, tolerance=1e-8, max_iterations=1000,
                    return_history=False):
    """Repeatedly multiply: r_{k+1} = M @ r_k until convergence.

    This is POWER ITERATION written out manually — no library function.
    If alpha is given, the matrix passed in is treated as P and G is built
    internally (this keeps the notebook cells readable).
    """
    n = M.shape[0]
    r = initial_rank(n) if r0 is None else r0.copy()
    if alpha is not None:
        M = google_matrix(M, alpha)
    history = [r.copy()]
    diff = np.inf
    k = 0
    while k < max_iterations:
        r_new = M @ r
        diff = np.linalg.norm(r_new - r, 1)   # L1 distance between iterates
        history.append(r_new.copy())
        r = r_new
        k += 1
        if diff < tolerance:
            break
    if return_history:
        return r, k, diff, history
    return r, k, diff


# =============================================================================
# STEPS 8 & 9 — verify the eigenvector equation, then compute eigenvectors
# =============================================================================
def verify_eigenvector(P, r):
    """Show P @ r ≈ r, i.e. P @ r = 1 * r (eigenvalue λ = 1)."""
    Pr = P @ r
    print("P @ r =", np.round(Pr, 6))
    print("r     =", np.round(r, 6))
    print("np.allclose(P @ r, r) →", np.allclose(P @ r, r))
    print("⇒  P @ r = 1 · r   (λ = 1)   ✓  r is an eigenvector of P")


def eigenvector_from_scipy(P):
    """Find the eigenvector for the eigenvalue closest to 1, normalized to
    sum to 1 (a probability vector)."""
    eigenvalues, eigenvectors = np.linalg.eig(P)
    i = int(np.argmin(np.abs(eigenvalues - 1.0)))
    v = np.real(eigenvectors[:, i])
    v = v / v.sum()                           # normalize: sum(r_i) = 1
    return eigenvalues, eigenvectors, v, i


# =============================================================================
# STEPS 10 & 11 — dangling nodes and damping (the Google matrix)
# =============================================================================
def google_matrix(P, alpha=0.85):
    """G = alpha * P + (1 - alpha) / n * J,  J = all-ones matrix.

    With probability alpha follow a link; with probability 1-alpha jump to a
    random page. This guarantees G is primitive (irreducible + aperiodic), so
    the iteration always converges to a unique ranking.
    """
    n = P.shape[0]
    return alpha * P + (1 - alpha) * np.ones((n, n)) / n


def dangling_demo():
    """A page with no outgoing links: watch probability leak without a fix."""
    # C has NO outgoing links
    pages = ["A", "B", "C"]
    links = [(0, 1), (1, 0), (0, 2)]          # A->B, B->A, A->C ; C is a dead end
    A = adjacency_matrix(3, links)

    P_naive = transition_matrix(A, fix_dangling=False)   # C's column = 0
    P_fixed = transition_matrix(A, fix_dangling=True)    # C's column = 1/3

    print("P_naive (C's column is 0 → probability leaks):\n", np.round(P_naive, 3))
    r_naive, k_naive, diff_naive = power_iteration(P_naive, tolerance=1e-8)
    print(f"after {k_naive} iterations the vector sums to {r_naive.sum():.4f} "
          f"(should be 1!) — rank has leaked away.\n")

    print("P_fixed (C redistributes uniformly):\n", np.round(P_fixed, 3))
    check_stochastic(P_fixed, "P_fixed")
    r_fixed, k_fixed, diff_fixed = power_iteration(P_fixed, tolerance=1e-8)
    print(f"after {k_fixed} iterations the vector sums to {r_fixed.sum():.4f} ✓\n")
    return P_naive, P_fixed

# =============================================================================
# STEP 15 — the reusable educational function
# =============================================================================
def pagerank(P, alpha=0.85, tolerance=1e-8, max_iterations=1000):
    """Educational PageRank: iterate r_{k+1} = G @ r_k until convergence.

    Returns (rank_vector, iteration_count, convergence_history).

    Parameters
    ----------
    P : 2D numpy array
        Column-stochastic transition matrix (dangling pages already handled).
    alpha : float
        Damping factor (probability of following a link).
    tolerance : float
        Stop when ||r_{k+1} - r_k||_1 < tolerance.
    max_iterations : int
        Safety limit.
    """
    n = P.shape[0]
    G = google_matrix(P, alpha)
    r = initial_rank(n)
    history = [r.copy()]
    k = 0
    diff = np.inf
    while k < max_iterations:
        r_new = G @ r
        diff = np.linalg.norm(r_new - r, 1)
        history.append(r_new.copy())
        r = r_new
        k += 1
        if diff < tolerance:
            break
    return r, k, history

# =============================================================================
# HOW DOES A MACHINE LEARN? — Calculus, Gradient Descent & Optimization Lab
# -----------------------------------------------------------------------------
# The *coding* companion to the web simulation. Every mathematical step is
# written out explicitly so you can see exactly what happens:
#
#   DATA → MODEL → PREDICTION → ERROR → LOSS → DERIVATIVE → GRADIENT
#   → GRADIENT DESCENT → OPTIMAL PARAMETERS
#
# We deliberately do NOT use sklearn.LinearRegression or automatic
# differentiation — you will implement the derivative yourself, first
# numerically (finite differences) and then analytically.
#
# CONVENTIONS (kept consistent everywhere in this file):
#   * data: arrays x (inputs) and y (targets)
#   * model:  ŷ = w·x + b
#   * loss:   L(w,b) = (1/n) · Σ (yᵢ - (w·xᵢ + b))²     (Mean Squared Error)
#   * partial derivatives:
#         ∂L/∂w = -(2/n) · Σ xᵢ·(yᵢ - ŷᵢ)
#         ∂L/∂b = -(2/n) · Σ (yᵢ - ŷᵢ)
#   * gradient:  ∇L = [∂L/∂w, ∂L/∂b]
#   * update:    θ ← θ - η·∇L          (gradient descent)
#
# Author: generated for the Calculus & Optimization course orientation
# =============================================================================

import numpy as np

try:
    import matplotlib.pyplot as plt
except ImportError:
    raise SystemExit("Install matplotlib first:  pip install matplotlib")


# =============================================================================
# STEP 2 — create the data (Hours Studied → Exam Score)
# =============================================================================
def hours_data():
    x = np.array([1, 2, 3, 4, 5, 6, 7, 8], dtype=float)
    y = np.array([35, 42, 51, 59, 68, 74, 83, 89], dtype=float)
    return x, y


def noisy_data(x, y, noise=6.0, seed=0):
    """Add uniform noise in [-noise, +noise] so perfect prediction is impossible."""
    rng = np.random.default_rng(seed)
    return x, y + rng.uniform(-noise, noise, size=y.shape)


def synthetic_data(n=16, seed=0):
    """A less-than-linear AI-flavored dataset to show the same method on harder data."""
    rng = np.random.default_rng(seed)
    x = np.linspace(-0.9, 0.9, n)
    y = 1.6 * x - 0.2 + 0.35 * np.sin(3 * x) + rng.uniform(-0.15, 0.15, n)
    return x, y


# =============================================================================
# STEP 3 — the model
# =============================================================================
def predict(x, w, b):
    """ŷ = w·x + b  — the model (a straight line)."""
    return w * x + b


# =============================================================================
# STEP 4 — errors / residuals
# =============================================================================
def errors(y, y_pred):
    """eᵢ = yᵢ - ŷᵢ — how far each prediction is from the truth."""
    return y - y_pred


# =============================================================================
# STEP 5 — the loss function (MSE)
# =============================================================================
def mse_loss(y, y_pred):
    """L = (1/n) Σ (yᵢ - ŷᵢ)²  — Mean Squared Error."""
    return np.mean((y - y_pred) ** 2)


# =============================================================================
# STEP 6 — loss as a function of w only (b frozen)
# =============================================================================
def loss_curve_1d(x, y, b_frozen=0.0, w_values=None):
    """L(w) for many w values, with b fixed. Returns (w_values, losses)."""
    if w_values is None:
        w_values = np.linspace(0, 14, 200)
    losses = np.array([mse_loss(y, predict(x, w, b_frozen)) for w in w_values])
    return w_values, losses


# =============================================================================
# STEP 7 — numerical derivative (finite differences) BEFORE the formula
# =============================================================================
def numerical_dLdw(x, y, w, b, h=1e-4):
    """Approximate dL/dw by (L(w+h) - L(w-h)) / (2h) — no calculus needed."""
    return (mse_loss(y, predict(x, w + h, b)) - mse_loss(y, predict(x, w - h, b))) / (2 * h)


# =============================================================================
# STEP 8 — analytical partial derivatives
# =============================================================================
def dLdw(x, y, w, b):
    """∂L/∂w = -(2/n) Σ xᵢ·(yᵢ - ŷᵢ)"""
    return -2 * np.sum(x * (y - predict(x, w, b))) / len(y)


def dLdb(x, y, w, b):
    """∂L/∂b = -(2/n) Σ (yᵢ - ŷᵢ)"""
    return -2 * np.sum(y - predict(x, w, b)) / len(y)


def gradient(x, y, w, b):
    """∇L = [∂L/∂w, ∂L/∂b] — the direction of steepest local INCREASE."""
    return np.array([dLdw(x, y, w, b), dLdb(x, y, w, b)])


# =============================================================================
# STEPS 9 & 10 — one gradient-descent step, then a full training loop
# =============================================================================
def one_step(x, y, w, b, learning_rate):
    """θ ← θ - η·∇L  — written component by component."""
    dw = dLdw(x, y, w, b)
    db = dLdb(x, y, w, b)
    w_new = w - learning_rate * dw
    b_new = b - learning_rate * db
    return w_new, b_new


def train(x, y, w0=0.0, b0=0.0, learning_rate=0.01, epochs=2000, tolerance=1e-6):
    """Run gradient descent; keep history of w, b, loss and gradient norms.

    Returns a dict with w, b, loss, epochs_run and histories (arrays).
    """
    w, b = w0, b0
    w_hist, b_hist, loss_hist, grad_hist = [], [], [], []
    for epoch in range(epochs):
        y_pred = predict(x, w, b)
        loss = mse_loss(y, y_pred)
        gw, gb = dLdw(x, y, w, b), dLdb(x, y, w, b)
        w_hist.append(w); b_hist.append(b); loss_hist.append(loss)
        grad_hist.append(np.hypot(gw, gb))
        w -= learning_rate * gw
        b -= learning_rate * gb
        if np.hypot(gw, gb) < tolerance:
            break
    y_pred = predict(x, w, b)
    return {"w": w, "b": b, "loss": mse_loss(y, y_pred),
            "epochs": len(loss_hist),
            "w_hist": np.array(w_hist), "b_hist": np.array(b_hist),
            "loss_hist": np.array(loss_hist), "grad_hist": np.array(grad_hist)}


# =============================================================================
# STEPS 13 — learning-rate experiment
# =============================================================================
def learning_rate_experiment(x, y, rates=(0.001, 0.01, 0.05, 0.2), epochs=200):
    """Run training at several learning rates and report the behaviour.

    Because the data has x ∈ [1,8], curvature is ~O(50): 0.01 converges
    smoothly, 0.05 overshoots/oscillates, 0.2 diverges. These are genuine
    numerical behaviours of THIS dataset, not hard-coded results.
    """
    print(f"{'eta':>7} {'final loss':>11} {'epochs':>7}  behaviour")
    results = {}
    for eta in rates:
        r = train(x, y, learning_rate=eta, epochs=epochs)
        results[eta] = r
        loss_0 = mse_loss(y, predict(x, 0, 0))
        loss_end = r["loss"]
        if loss_end > loss_0 * 10:
            beh = "DIVERGES"
        elif np.any(np.diff(r["loss_hist"]) > 0):
            beh = "overshoots/oscillates"
        else:
            beh = "smooth convergence"
        print(f"{eta:7.3f} {loss_end:11.2f} {r['epochs']:7d}  {beh}")
    return results


# =============================================================================
# STEPS 14 & 15 — loss surface grid + trajectory
# =============================================================================
def loss_grid(x, y, w_range, b_range, n=40):
    """L(w,b) on a grid — used for both the 3D surface and the contour map."""
    W = np.linspace(*w_range, n)
    B = np.linspace(*b_range, n)
    WW, BB = np.meshgrid(W, B)
    L = np.zeros_like(WW)
    for i in range(n):
        for j in range(n):
            L[i, j] = mse_loss(y, predict(x, WW[i, j], BB[i, j]))
    return W, B, L


def plot_surface(x, y, result, w_range=(-2, 14), b_range=(-10, 110), n=40):
    """3D loss surface with the gradient-descent trajectory drawn on it."""
    from mpl_toolkits.mplot3d import Axes3D  # noqa: F401 (registers the 3d projection)
    W, B, L = loss_grid(x, y, w_range, b_range, n)
    fig = plt.figure(figsize=(10, 7))
    ax = fig.add_subplot(111, projection="3d")
    ax.plot_surface(W, B, L, cmap="viridis", alpha=0.85, linewidth=0, antialiased=True)
    ax.plot(result["w_hist"], result["b_hist"], result["loss_hist"],
            color="#fbbf24", lw=2.5, marker="o", markersize=3,
            label="gradient descent path")
    ax.scatter(result["w_hist"][0], result["b_hist"][0], result["loss_hist"][0],
               color="#fbbf24", s=90, marker="*")
    ax.scatter(result["w"], result["b"], result["loss"], color="#22d3ee", s=110)
    ax.set_xlabel("w"); ax.set_ylabel("b"); ax.set_zlabel("L(w,b)")
    ax.set_title("Loss surface L(w,b) with the gradient-descent trajectory")
    ax.legend()
    plt.tight_layout()
    plt.show()


def plot_contour(x, y, result, w_range=(-2, 14), b_range=(-10, 110), n=80):
    """Contour map of L(w,b) with the trajectory overlaid (direction is key)."""
    W, B, L = loss_grid(x, y, w_range, b_range, n)
    plt.figure(figsize=(9, 7))
    cf = plt.contourf(W, B, L, levels=20, cmap="viridis")
    cs = plt.contour(W, B, L, levels=15, colors="w", alpha=0.4)
    plt.clabel(cs, inline=True, fontsize=8, fmt="%.0f")
    plt.plot(result["w_hist"], result["b_hist"], color="#fbbf24", lw=2.5,
             marker="o", markersize=3, label="trajectory")
    plt.scatter(result["w_hist"][0], result["b_hist"][0], color="#fbbf24", s=90, marker="*")
    plt.scatter(result["w"], result["b"], color="#22d3ee", s=110, zorder=5)
    plt.colorbar(cf, label="L(w,b)")
    plt.xlabel("w"); plt.ylabel("b")
    plt.title("Contour map of L(w,b) — the trajectory follows −∇L")
    plt.legend()
    plt.tight_layout()
    plt.show()


# =============================================================================
# STEPS 16 & 17 — matrix/vector form  ŷ = X·θ  and  L = (1/n)‖y − Xθ‖²
# =============================================================================
def design_matrix(x):
    """X = [[1, x₁], [1, x₂], ..., [1, xₙ]] — the column of 1s pairs with b."""
    return np.column_stack([np.ones_like(x), x])


def predict_matrix(X, theta):
    """ŷ = X·θ with θ = [b, w]. Same numbers as w·x + b."""
    return X @ theta


def mse_loss_matrix(y, X, theta):
    """L(θ) = (1/n) ‖y − Xθ‖²₂  — the norm view of the same loss."""
    residual = y - X @ theta
    return np.linalg.norm(residual) ** 2 / len(y)


# =============================================================================
# STEP 18 — normal equation (least squares, for comparison only)
# =============================================================================
def normal_equation(X, y):
    """θ̂ = (XᵀX)⁻¹Xᵀy — closed form, when XᵀX is invertible (use lstsq for safety)."""
    return np.linalg.solve(X.T @ X, X.T @ y)


def least_squares(X, y):
    """Numerically safer closed form via SVD-based least squares."""
    return np.linalg.lstsq(X, y, rcond=None)[0]
